cmake_minimum_required(VERSION 3.22)
project(il2cppdumper_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DUMPER_SANITIZERS "Run portable tests with ASan and UBSan" OFF)
enable_testing()
add_library(dumper_core STATIC module/src/main/cpp/core/elf.cpp module/src/main/cpp/core/maps.cpp
    module/src/main/cpp/core/runtime_layout.cpp module/src/main/cpp/core/target.cpp module/src/main/cpp/core/output.cpp)
target_include_directories(dumper_core PUBLIC module/src/main/cpp)
target_compile_options(dumper_core PRIVATE -Wall -Wextra -Wpedantic -Wformat=2 -Wshadow -Werror)
if(DUMPER_SANITIZERS)
    target_compile_options(dumper_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(dumper_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE dumper_core)
target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME core_tests COMMAND core_tests)
add_executable(elf_tests tests/elf_tests.cpp)
target_link_libraries(elf_tests PRIVATE dumper_core)
target_compile_options(elf_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME elf_tests COMMAND elf_tests)
