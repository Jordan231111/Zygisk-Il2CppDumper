cmake_minimum_required(VERSION 3.22)
project(il2cppdumper_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(DUMPER_SANITIZERS "Run portable tests with ASan and UBSan" OFF)
enable_testing()
add_library(dumper_core STATIC module/src/main/cpp/core/elf.cpp module/src/main/cpp/core/maps.cpp
    module/src/main/cpp/core/runtime_layout.cpp module/src/main/cpp/core/target.cpp module/src/main/cpp/core/output.cpp)
target_include_directories(dumper_core PUBLIC module/src/main/cpp)
target_compile_options(dumper_core PRIVATE -Wall -Wextra -Wpedantic -Wformat=2 -Wshadow -Werror)
if(DUMPER_SANITIZERS)
    target_compile_options(dumper_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(dumper_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE dumper_core)
target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME core_tests COMMAND core_tests)
add_executable(elf_tests tests/elf_tests.cpp)
target_link_libraries(elf_tests PRIVATE dumper_core)
target_compile_options(elf_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME elf_tests COMMAND elf_tests)

if(ANDROID)
    enable_language(C)
    set(native_dir module/src/main/cpp)
    add_library(test_xdl STATIC ${native_dir}/xdl/xdl.c ${native_dir}/xdl/xdl_iterate.c
        ${native_dir}/xdl/xdl_linker.c ${native_dir}/xdl/xdl_lzma.c ${native_dir}/xdl/xdl_util.c)
    target_include_directories(test_xdl PUBLIC ${native_dir}/xdl/include)
    foreach(fixture IN ITEMS il2cpp renamed_runtime hidden_runtime missing_runtime
            capabilities_runtime calibration_runtime)
        add_library(${fixture} SHARED tests/mock_il2cpp.cpp)
        target_include_directories(${fixture} PRIVATE ${native_dir})
        target_compile_options(${fixture} PRIVATE -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden)
        target_link_options(${fixture} PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
    endforeach()
    target_compile_definitions(renamed_runtime PRIVATE FIXTURE_LEGACY=1 FIXTURE_LAYOUT_WORDS=2)
    target_compile_definitions(hidden_runtime PRIVATE FIXTURE_HIDE_CORE=1)
    target_compile_definitions(missing_runtime PRIVATE FIXTURE_OMIT_CORE=1)
    target_compile_definitions(capabilities_runtime PRIVATE FIXTURE_HIDE_CAPABILITIES=1)
    target_compile_definitions(calibration_runtime PRIVATE FIXTURE_HIDE_CALIBRATION=1)
    add_executable(runtime_tests tests/runtime_tests.cpp ${native_dir}/il2cpp_api.cpp
        ${native_dir}/il2cpp_dump.cpp ${native_dir}/discovery.cpp)
    target_link_libraries(runtime_tests PRIVATE dumper_core test_xdl log dl)
    target_compile_options(runtime_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

option(DUMPER_BENCHMARKS "Build the isolated output-staging benchmark" OFF)
if(DUMPER_BENCHMARKS)
    add_executable(output_benchmark tests/output_benchmark.cpp)
    target_link_libraries(output_benchmark PRIVATE dumper_core)
endif()
